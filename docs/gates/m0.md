## 4. 交付的公共 API 契约

```go
// 错误
ErrAgain  ErrEOF(= io.EOF)  ErrInvalidData  ErrUnsupported  ErrResourceLimit
IsFatal(err) bool

// 帧
type PixelFormat uint8            // 6 种：420/422/444 × 8/10 bit
(f PixelFormat) String/Valid/BitDepth/BytesPerSample/ChromaSubsampling/PlaneDimensions/RowBytes
type Plane struct { Data []byte; Stride int }
type Frame struct { Width, Height, Format, Planes[3], PTS, Duration, KeyFrame }
(*Frame) Validate() error
(*Frame) Clone() *Frame
(*Frame) Release()

// 包
type Packet struct { Data, PTS, DTS, Duration, KeyFrame }
(Packet) Valid() bool
(Packet) Clone() Packet

// 配置
type RateControl uint8            // CQP / ABR / CBR / VBV
type EncoderConfig struct { ... } // SetDefaults() / Validate()
type DecoderConfig struct { ... } // SetDefaults() / Validate()
```
